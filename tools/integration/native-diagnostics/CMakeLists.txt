cmake_minimum_required(VERSION 3.15)
project(rmsim_native_diagnostics LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(diagnostic_updater REQUIRED)
add_executable(diagnostics_probe probe.cpp)
target_compile_features(diagnostics_probe PRIVATE cxx_std_17)
ament_target_dependencies(diagnostics_probe rclcpp diagnostic_updater)
target_link_libraries(diagnostics_probe dbghelp)
target_compile_options(diagnostics_probe PRIVATE /Zi /Od /EHsc)
target_link_options(diagnostics_probe PRIVATE /DEBUG)
